import tarfile, zipfile, json, pathlib, posixpath, stat, hashlib, zlib, base64
root=pathlib.Path('/tmp/orca639')
file=root/'package/ORCA-0.1.0-linux-x64.tar.gz'
def hash_file(p):
 h=hashlib.sha512()
 with open(p,'rb') as f:
  for chunk in iter(lambda:f.read(1024*1024),b''):h.update(chunk)
 return h.hexdigest()
critical={};symlinks=[]
with tarfile.open(file,'r:gz') as tar:
 members=tar.getmembers()
 for m in members:
  assert not m.name.startswith('/') and '..' not in pathlib.PurePosixPath(m.name).parts,m.name
  if m.issym() or m.islnk():
   target=posixpath.normpath(posixpath.join(posixpath.dirname(m.name),m.linkname))
   assert not target.startswith('/') and '..' not in pathlib.PurePosixPath(target).parts,(m.name,m.linkname)
   symlinks.append({'name':m.name,'target':m.linkname})
  for suffix in ['resources/app.asar','resources/daemon/orca-daemon.mjs','orca']:
   if m.isfile() and m.name.endswith('/'+suffix):
    expected=root/'package/linux-unpacked'/suffix
    stream=tar.extractfile(m);h=hashlib.sha256()
    for chunk in iter(lambda:stream.read(1024*1024),b''):h.update(chunk)
    assert h.hexdigest()==hashlib.sha256(expected.read_bytes()).hexdigest(),suffix
    critical[suffix]={'size':m.size,'mode':oct(m.mode),'sha256':h.hexdigest()}
assert len(critical)==3
def validate_chunks(data, blockmap):
 offset=0
 for file in blockmap['files']:
  assert file['offset']==0
  for size,expected in zip(file['sizes'],file['checksums'],strict=True):
   actual=base64.b64encode(hashlib.blake2b(data[offset:offset+size],digest_size=18).digest()).decode()
   assert actual==expected,(offset,size)
   offset+=size
 assert offset==len(data)

assert next(r for r in archives['results'] if r['name']=='web-stored')['blockMapSize']>0
